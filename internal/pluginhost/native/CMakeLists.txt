cmake_minimum_required(VERSION 3.25)
project(IceTudioPluginScanner LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The IceTudio plugin scanner currently targets Linux")
endif()
set(CLAP_ROOT "" CACHE PATH "Official CLAP SDK checkout")
set(VST3_ROOT "" CACHE PATH "Official VST3 SDK checkout with hosting submodules")
if(NOT EXISTS "${CLAP_ROOT}/include/clap/clap.h" OR NOT EXISTS "${VST3_ROOT}/public.sdk/source/vst/hosting/module.h")
    message(FATAL_ERROR "Set CLAP_ROOT and VST3_ROOT to the pinned SDK checkouts")
endif()
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
add_subdirectory("${VST3_ROOT}" vst3 EXCLUDE_FROM_ALL)

add_executable(icetudio-plugin-scan scan.cpp "${VST3_ROOT}/public.sdk/source/vst/hosting/module_linux.cpp")
target_include_directories(icetudio-plugin-scan PRIVATE "${CLAP_ROOT}/include")
target_compile_features(icetudio-plugin-scan PRIVATE cxx_std_20)
target_link_libraries(icetudio-plugin-scan PRIVATE sdk_hosting ${CMAKE_DL_LIBS})
